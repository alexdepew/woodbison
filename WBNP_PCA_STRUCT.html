  • Revised_cleandata_07.03.2026.csv

Load Required Data

cleaned.data <- read.csv("./datafiles/Revised_cleandata_07.03.2026.csv",
                         check.names = F)

Convert data to Genind object

# Make genind object
bison.genind <- df2genind(X=cleaned.data[,c(19:ncol(cleaned.data))],
                              sep = "/",
                              ind.names = cleaned.data$Spreadsheet_ID,
                              loc.names = NULL,
                              pop = cleaned.data$HERD,
                              NA.char = "NA",
                              ploidy = 2,
                              type = "codom",
                              strata = NULL,
                              hierarchy = NULL)

# Attach collection year metadata
idx <- match(indNames(bison.genind), cleaned.data$Spreadsheet_ID)
stopifnot(!any(is.na(idx)))
bison.genind@other$YEAR.ASSIGN <- cleaned.data$YEAR.ASSIGN[idx]

Subset data

Data is subset into two temporal groups. Temporal groups are: 1: pre-2016, and 3: post-2016

# Create datasets containing samples of interest for temporal comparisons
w1 <- bison.genind[bison.genind@pop %in% c("WBNPGR", "WBNPHC", 
                                           "WBNPLB", "WBNPNY", "WBNPPAD", "WENT") &
                   bison.genind@other$YEAR.ASSIGN <=2015,]

w2 <- bison.genind[bison.genind@pop %in% c("WBNPGR", "WBNPHC", 
                                           "WBNPLB", "WBNPNY", "WBNPPAD", "WENT") &
                   bison.genind@other$YEAR.ASSIGN >=2016,]

PCA

Perform PCA on both datasets

# Perform PCA
x.w1 <- tab(w1, freq = TRUE, NA.method = "mean")
pca.w1 <- dudi.pca(x.w1, center = T, scale = F, scannf = F, nf = 2)

# Axes % Variance
eig.perc.w1 <- 100*pca.w1$eig/sum(pca.w1$eig)

x.w2 <- tab(w2, freq = TRUE, NA.method = "mean")
pca.w2 <- dudi.pca(x.w2, center = T, scale = F, scannf = F, nf = 2)

# Axes % Variance
eig.perc.w2 <- 100*pca.w2$eig/sum(pca.w2$eig)

#### axis limits
xlims <- c(-2.5, 3.5)
ylims <- c(-2, 2.1)

PCA plot with ggplot

WBNP Temp1 (W1: No EINP)

w1@pop <- factor(w1@pop, levels = sort(unique(as.character(w1@pop))))
levels_w1 <- levels(w1@pop)
w1_cols <- wbnp_cols[levels_w1]


# Plot PCA object
pcaw1 <- fviz_pca_ind(pca.w1,
             axes = c(1, 2),
             geom = "blank",
             habillage = w1@pop,
             repel = TRUE,
             palette = w1_cols,
             alpha.ind = 1,) + 
  stat_ellipse(aes(color = w1@pop), type = "norm", level = 0.68) + 
  geom_point(aes(colour = w1@pop),
             shape = 19,
             alpha = 0.75,
             size = 1.2) + 
coord_fixed(ratio = 1, xlim = xlims, ylim = ylims) +
  
theme(
    plot.title = element_blank(),
    legend.title = element_text(face = "bold"), # Kept bold but removed blank
    axis.title = element_text(face = "bold"),
    panel.border = element_rect(colour = "black", fill = NA, linewidth = 1),  ) + 

  labs(title = NULL,
       # Fixed the placement of closing parentheses and commas below
       x = paste0("PC1 (", round(eig.perc.w1[1], 2), "%)"),
       y = paste0("PC2 (", round(eig.perc.w1[2], 2), "%)"),

    guides(colour = guide_legend(override.aes = list(size = 4, shape = 19)))
